        if (result.getResultCode() == RESULT_OK) {
            Intent data = result.getData();
            if (data != null){
                boolean status = data.getBooleanExtra("status", false);
                int response = data.getIntExtra("response", 0);
                String message = data.getStringExtra("message");
                String dataObject = data.getStringExtra("data");

                String detailedResponse = "Status: " + status + ", Response: " + response + ", Message: " + message + ", Data: " + dataObject;
                showSnack(detailedResponse); // Replace with your snack method
                Log.i("casa", detailedResponse);
            }
        }
